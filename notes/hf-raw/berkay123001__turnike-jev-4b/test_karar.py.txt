def _bosalt() -> None:
    _DURUM.pop("karar", None)
    gc.collect()
    if torch.cuda.is_available():
        torch.cuda.empty_cache()


def karar() -> Karar:
    if "karar" not in _DURUM:
        _DURUM["karar"] = Karar(ADAPTER, TABAN)
    return _DURUM["karar"]


def bf16_sonuclari() -> dict[str, dict[str, list[float]]]:
    """9 ornegin bf16 olasiliklari (bir kez hesaplanir, referans ve FP8 testleri de kullanir)."""
    if "bf16" not in _DURUM:
        k = karar()
        _DURUM["bf16"] = {ad: k.olasiliklar(metin, sorular) for ad, metin, sorular, _ in ORNEKLER}
    return _DURUM["bf16"]


def _en_buyuk_fark(a: dict[str, dict[str, list[float]]], b: dict[str, dict[str, list[float]]]) -> float:
    return max(abs(x - y) for ad in a for s in a[ad] for x, y in zip(a[ad][s], b[ad][s]))


def _argmax_uyumu(a, b) -> tuple[int, int]:
    """(karsilastirilan, uyusan): yalniz a'da en iyi iki secenek ARGMAX_PAYI'ndan acik olan sorular."""
    toplam = uyan = 0
    for ad in a:
        for s, pa in a[ad].items():
            ilk_iki = sorted(pa, reverse=True)[:2]
            if ilk_iki[0] - ilk_iki[1] < ARGMAX_PAYI:
                continue
            toplam += 1
            uyan += int(pa.index(max(pa)) == b[ad][s].index(max(b[ad][s])))
    return toplam, uyan

def test_bagimsiz_yukleme() -> None:
    """karar.py yalniz stdlib + torch/transformers/peft/torchao import eder ve tek basina kopyalanip yuklenir."""
    agac = ast.parse((BURASI / "karar.py").read_text(encoding="utf-8"))
    kokler = {a.name.split(".")[0] for d in ast.walk(agac) if isinstance(d, ast.Import) for a in d.names}
    kokler |= {d.module.split(".")[0] for d in ast.walk(agac) if isinstance(d, ast.ImportFrom) and d.module}
    yabanci = kokler - set(sys.stdlib_module_names) - IZINLI_UCUNCU_PARTI - {"__future__"}

def test_hatali_girdi() -> None:
    """Hatali her bicim ValueError; dogru bicimler kabul, alan adlari ve noul etiketleri dogru."""
    kotuler = [("", {"a?": ["x", "y"]}), ("  ", {"a?": ["x", "y"]}), (None, {"a?": ["x", "y"]}),
               ("m", {}), ("m", [("a?", ["x", "y"])]), ("m", {"a?": ["x"]}), ("m", {"a?": ["x", "x"]}),
               ("m", {"a?": "evet"}), ("m", {"a?": ["x", " "]}), ("m", {"": ["x", "y"]}), ("m", {"a?": 3}),
               ("m", {"a?": {"type": "bilinmeyen", "criteria": ["x", "y"]}}),
               ("m", {"a?": {"type": "choice", "criteria": "x"}}), ("m", {"a?": {"type": "choice"}}),
               ("m", {"a?": {"type": "noul", "criteria": ["evet", "hayır"]}}),
               ("m", {"a <decision> b": ["x", "y"]}), ("m", {"a?": {"x": ""}}),
               ("m", {"a?": {"type": "choice", "criteria": {"x": "y"}}}),
               ("m", {"a?": {"type": "noul", "instruction": "yazim hatasi"}}),
               ("m", {"a": {"type": "noul", "instructions": "x <decision> y"}}),
               ("m", {"a?": ["x", "y <decision>"]}), ("m", {"a?": [str(i) for i in range(len(SEMBOLLER) ** 2 + 1)]})]
    for metin, sorular in kotuler:
        try:
            girdi_dogrula(metin, sorular)
        except ValueError:
            continue
        raise AssertionError(f"ValueError bekleniyordu: {metin!r} {sorular!r}")
    tek = girdi_dogrula("m", {"Ton nedir?": ("a", "b")})
    assert tek == [("Ton nedir?", "q", {"type": "choice", "instructions": "Ton nedir?", "criteria": {"a": "a", "b": "b"}})]
    iki = girdi_dogrula("m", {"ton": ["a", "b"], "Acil mi?": {"type": "noul"}})
    assert [ad for _, ad, _ in iki] == ["q1", "q2"]
    assert [ad for _, ad, _ in girdi_dogrula("m", {"ton": ["a", "b"], "acil": {"type": "noul"}})] == ["ton", "acil"]
    assert [ad for _, ad, _ in girdi_dogrula("m", {"ton\n": ["a", "b"]})] == ["q"]  # sonda satir sonu: alan adi degil
    assert secenekler({"type": "noul"})[0] == ["false", "true"]
    print(f"{len(kotuler)} hatali girdi reddedildi")

def test_dokuz_ornek() -> None:
    """9 ornek (TR/EN; ikili, coktan secmeli, puan, 'hicbiri'li, cok soru paketli): bicim + acik orneklerde argmax."""
    k = karar()
    basla = time.perf_counter()
    sonuc = bf16_sonuclari()
    print(f"9 ornek: {(time.perf_counter() - basla) * 1000:.0f} ms toplam (ilk cagri isinma dahil)")
    for ad, metin, sorular, beklenen in ORNEKLER:
        dagilimlar = k.sor(metin, sorular)
        assert list(dagilimlar) == list(sorular)
        for soru, dagilim in dagilimlar.items():
            assert list(dagilim) == secenekler(girdi_dogrula(metin, {soru: sorular[soru]})[0][2])[0]
            assert math.isclose(sum(dagilim.values()), 1.0, abs_tol=1e-4)
            assert [round(p, 6) for p in dagilim.values()] == [round(p, 6) for p in sonuc[ad][soru]]
            en_iyi = max(dagilim, key=dagilim.get)
            print(f"  {ad:18s} {soru[:48]:48s} -> {en_iyi} ({dagilim[en_iyi]:.3f})")
            if beklenen and soru in beklenen:
                assert en_iyi == beklenen[soru], f"{ad}: {soru} -> {en_iyi}, beklenen {beklenen[soru]}"

def test_uzun_metin_cok_secenek_cok_soru() -> None:
    """Sinir asan metin sondan kesilir; 62+ sik iki asamali eleme; 16+ soru dengeli paketler."""
    k = karar()
    eski, k.maks_istem_token = k.maks_istem_token, 512
    try:
        once = k.kesilen
        uzun = k.sor("\n\n".join([EPOSTA] * 30), {"Mesajın tonu nedir?": ["olumlu", "olumsuz", "nötr"]})
        assert k.kesilen == once + 1 and math.isclose(sum(uzun["Mesajın tonu nedir?"].values()), 1, abs_tol=1e-4)
    finally:
        k.maks_istem_token = eski
    niyetler = [f"niyet {i:02d}" for i in range(70)] + ["kart iptali"]
    cok = k.sor("Kartımı kaybettim, lütfen hemen iptal edin.", {"Müşterinin niyeti nedir?": niyetler})
    p = cok["Müşterinin niyeti nedir?"]
    sifir_disi = sum(v > 0 for v in p.values())
    print(f"71 sik: sifir disi {sifir_disi}, en iyi {max(p, key=p.get)} ({max(p.values()):.3f})")
    assert len(p) == 71 and sifir_disi <= len(SEMBOLLER) and math.isclose(sum(p.values()), 1, abs_tol=1e-4)
    on_sekiz = {f"s{i}": {"type": "noul", "instructions": f"Metinde {i} sayısı geçiyor mu?"} for i in range(18)}
    paket = k.sor("Siparişim 3 gün gecikti, 7 numaralı şubeyi aradım.", on_sekiz)
    assert list(paket) == list(on_sekiz) and len(on_sekiz) > MAKS_SORU
    print(f"18 soru -> {math.ceil(18 / MAKS_SORU)} gecis; s3 evet {paket['s3']['true']:.3f}, s5 evet {paket['s5']['true']:.3f}")


# --- 8: referans (olcum kodu) ---------------------------------------------------
def test_referans_esdeger() -> None:
    """Ayni girdi: karar.py vs olcum kodundaki Qwen35Decider (kendi yuklemesiyle). max |dp| <= REFERANS_TOLERANS."""
    if not REFERANS:
        print("ATLANDI: KARAR_REFERANS verilmedi")
        _DURUM["atlanan"] = _DURUM.get("atlanan", 0) + 1
        return
    bizim = dict(bf16_sonuclari())
    k = karar()
    bizim.update({ad: k.olasiliklar(metin, sorular) for ad, metin, sorular in EK_REFERANS})
    del k  # yerel referans kalirsa model bellekten cikmaz
    _bosalt()
    sys.path.insert(0, REFERANS)
    from qwen35_decider import Qwen35Decider  # noqa: E402 (yalniz bu testte, yerel olcum kodu)

    ref = Qwen35Decider(TABAN, adapter=ADAPTER)
    try:
        onlarin = {}
        for ad, metin, sorular in [o[:3] for o in ORNEKLER] + EK_REFERANS:
            temiz = girdi_dogrula(metin, sorular)
            istekler = {alan: istek for _, alan, istek in temiz}
            olas = ref._olasiliklar(metin, list(istekler), istekler)
            onlarin[ad] = {anahtar: olas[alan] for anahtar, alan, _ in temiz}
    finally:
        del ref
        gc.collect()
        torch.cuda.empty_cache() if torch.cuda.is_available() else None
    fark = _en_buyuk_fark(bizim, onlarin)
    toplam, uyan = _argmax_uyumu(bizim, onlarin)
    _DURUM["referans_fark"] = fark
    print(f"referans: max |dp| {fark:.5f} (sinir {REFERANS_TOLERANS}), argmax {uyan}/{toplam}")
    assert fark <= REFERANS_TOLERANS and uyan == toplam

def test_fp8() -> None:
    """fp8=True yukler ve calisir; acik sorularda argmax bf16 ile ayni, |dp| gevsek sinirda."""
    if not torch.cuda.is_available():
        print("ATLANDI: CUDA yok")
        _DURUM["atlanan"] = _DURUM.get("atlanan", 0) + 1
        return
    bizim = bf16_sonuclari()
    _bosalt()
    k8 = Karar(ADAPTER, TABAN, fp8=True)
    try:
        fp8 = {ad: k8.olasiliklar(metin, sorular) for ad, metin, sorular, _ in ORNEKLER}
        basla = time.perf_counter()
        k8.olasiliklar(EPOSTA, ORNEKLER[3][2])
        torch.cuda.synchronize()
        ms = (time.perf_counter() - basla) * 1000
    finally:
        del k8
        gc.collect()
        torch.cuda.empty_cache()
    fark = _en_buyuk_fark(bizim, fp8)
    toplam, uyan = _argmax_uyumu(bizim, fp8)
    _DURUM["fp8_fark"] = fark
    _DURUM["fp8"] = fp8
    print(f"FP8: max |dp| {fark:.4f} (sinir {FP8_TOLERANS}), argmax {uyan}/{toplam}, 4 soru {ms:.0f} ms")
    assert fark <= FP8_TOLERANS and uyan == toplam


# --- 10: birlesik agirlik + yukleme aninda FP8 ------------------------------------
def test_birlesik_fp8_akis() -> None:
    """Karar(None, taban=BIRLESIK, fp8=True): katmanlar yuklenirken FP8 (TorchAoConfig). Sonradan FP8 ile ayni olasiliklar
    (|dp| <= 1e-3), bf16 ile argmax uyumu; bf16 kopyasi olusmadigi icin tepe bellek bf16 yuklemesinden kucuk."""
    if not BIRLESIK or not torch.cuda.is_available():
        print("ATLANDI: KARAR_BIRLESIK verilmedi" if not BIRLESIK else "ATLANDI: CUDA yok")
        _DURUM["atlanan"] = _DURUM.get("atlanan", 0) + 1
        return
    bizim = bf16_sonuclari()
    _bosalt()
    torch.cuda.reset_peak_memory_stats()
    k8 = Karar(None, BIRLESIK, fp8=True)
    try:
        tepe = torch.cuda.max_memory_allocated() / 1e9
        kuantize = sum(1 for _, m in k8.model.named_modules()
                       if isinstance(m, torch.nn.Linear) and type(m.weight).__name__ != "Parameter")
        akis = {ad: k8.olasiliklar(metin, sorular) for ad, metin, sorular, _ in ORNEKLER}
    finally:
        del k8
        gc.collect()
        torch.cuda.empty_cache()
    fark_bf = _en_buyuk_fark(bizim, akis)
    toplam, uyan = _argmax_uyumu(bizim, akis)
    fark_fp8 = _en_buyuk_fark(_DURUM["fp8"], akis) if "fp8" in _DURUM else None
    print(f"birlesik + akis FP8: kuantize lineer {kuantize}, tepe {tepe:.2f} GB, vs bf16 max |dp| {fark_bf:.4f} argmax "
          f"{uyan}/{toplam}, vs sonradan FP8 max |dp| {'-' if fark_fp8 is None else f'{fark_fp8:.5f}'}")
    assert kuantize > 0 and fark_bf <= FP8_TOLERANS and uyan == toplam
    assert fark_fp8 is None or fark_fp8 <= 1e-3
